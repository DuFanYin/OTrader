# core::MarketData (IB, simulated); data_loader (a day's files: Parquet, the cache) and
# historical_data (a backtest day in memory, and the replay days are rolled into).
if(BUILD_GATEWAY)
  add_library(ib_market_data ib_market_data.cpp)
  target_link_libraries(ib_market_data PUBLIC market_data_engine PRIVATE ib_api)
endif()
if(BUILD_BACKTEST)
  # Ubuntu's Arrow config would look for lz4's, which it does not ship.
  set(CMAKE_DISABLE_FIND_PACKAGE_lz4 TRUE)
  find_package(Arrow REQUIRED GLOBAL)
  find_package(Parquet REQUIRED GLOBAL)
  foreach(lib Arrow::arrow Parquet::parquet)
    if(NOT TARGET ${lib})
      if(TARGET ${lib}_shared)
        add_library(${lib} INTERFACE IMPORTED GLOBAL)
        target_link_libraries(${lib} INTERFACE ${lib}_shared)
      elseif(TARGET ${lib}_static)
        add_library(${lib} INTERFACE IMPORTED GLOBAL)
        target_link_libraries(${lib} INTERFACE ${lib}_static)
      else()
        message(FATAL_ERROR "${lib} not found (nor ${lib}_shared, ${lib}_static)")
      endif()
    endif()
  endforeach()

  add_library(data_loader data_loader.cpp)
  target_link_libraries(data_loader PUBLIC model PRIVATE Arrow::arrow Parquet::parquet
                        nlohmann_json::nlohmann_json)
  add_library(historical_data historical_data.cpp)
  target_link_libraries(historical_data PUBLIC data_loader model PRIVATE pricing)
endif()
if(OPTRA_SIMULATED_MARKET)
  add_library(simulated_market_data simulated_market_data.cpp)
  target_link_libraries(simulated_market_data PUBLIC market_data_engine PRIVATE data_loader)
endif()
