# ib_common: what the IB order gateway (infra/gateway) and the IB market data feed
# (infra/marketdata/feed_ib) share: a session on IB Gateway / TWS, the settings from the
# environment, and IB contracts in the platform's terms. Built with BUILD_GATEWAY only.
cmake_minimum_required(VERSION 3.16)
project(ib_common LANGUAGES CXX)


find_package(Threads REQUIRED)  # Protobuf: found by the top-level CMakeLists

# The TWS API (IBJts, GPL-3.0) and the Intel decimal library it needs are not in git:
# scripts/fetch_ibjts.sh puts them in thirdparty/IBJts and thirdparty/IntelRDFP. The API is
# compiled here, with its messages generated by this build's protoc (the .pb files it ships are
# for another protobuf version).
set(THIRDPARTY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../thirdparty")
set(TWS_CLIENT_DIR "${THIRDPARTY_DIR}/IBJts/source/cppclient/client")
set(TWS_PROTO_DIR "${THIRDPARTY_DIR}/IBJts/source/proto")
set(INTEL_BID_LIB "${THIRDPARTY_DIR}/IntelRDFP/libbid.a")
if(NOT EXISTS "${TWS_CLIENT_DIR}/EClientSocket.h" OR NOT EXISTS "${INTEL_BID_LIB}")
  message(FATAL_ERROR "BUILD_GATEWAY=ON needs the TWS API and libbid in ${THIRDPARTY_DIR}; run scripts/fetch_ibjts.sh")
endif()

find_program(PROTOC NAMES protoc REQUIRED)
file(GLOB TWS_PROTOS "${TWS_PROTO_DIR}/*.proto")
set(TWS_PROTO_GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/twsapi_proto")
set(TWS_PROTO_OUT)
foreach(proto IN LISTS TWS_PROTOS)
  get_filename_component(name "${proto}" NAME_WE)
  list(APPEND TWS_PROTO_OUT "${TWS_PROTO_GEN_DIR}/${name}.pb.cc" "${TWS_PROTO_GEN_DIR}/${name}.pb.h")
endforeach()
file(MAKE_DIRECTORY "${TWS_PROTO_GEN_DIR}")
add_custom_command(
  OUTPUT ${TWS_PROTO_OUT}
  COMMAND ${PROTOC} -I "${TWS_PROTO_DIR}" --cpp_out "${TWS_PROTO_GEN_DIR}" ${TWS_PROTOS}
  DEPENDS ${TWS_PROTOS}
  COMMENT "protoc: the TWS API's messages"
  VERBATIM)

file(GLOB TWS_SOURCES "${TWS_CLIENT_DIR}/*.cpp")
add_library(twsapi STATIC ${TWS_SOURCES} ${TWS_PROTO_OUT})
set_target_properties(twsapi PROPERTIES CXX_STANDARD 17)  # its own standard, not the engine's
# SYSTEM: the engine's warnings stay about the engine's code.
target_include_directories(twsapi SYSTEM PUBLIC "${TWS_CLIENT_DIR}" "${TWS_PROTO_GEN_DIR}")
target_link_libraries(twsapi PUBLIC protobuf::libprotobuf "${INTEL_BID_LIB}" Threads::Threads)

add_library(ib_common ib_session.cpp ib_settings.cpp ib_contracts.cpp)
target_include_directories(ib_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ib_common PUBLIC twsapi model)

# IbGateway: the live venue with IB. PUBLIC: the live assembly picks it where it is built.
find_package(Threads REQUIRED)
add_library(gateway_core engine_gateway_ib.cpp)
target_compile_definitions(gateway_core PUBLIC OPTRA_WITH_IB)
target_link_libraries(gateway_core PUBLIC gateway PRIVATE ib_common Threads::Threads)

# IbFeed: IB quotes, on a connection of their own, for the live market data.
add_library(ib_feed feed_ib.cpp)
target_link_libraries(ib_feed PUBLIC marketdata_live PRIVATE ib_common)
